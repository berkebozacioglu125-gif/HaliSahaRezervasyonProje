
CREATE DATABASE IF NOT EXISTS halisahadb;
USE halisahadb;


DROP TABLE IF EXISTS rezervasyonlar;
DROP TABLE IF EXISTS sahalar;
DROP TABLE IF EXISTS kullanicilar;


CREATE TABLE kullanicilar (
    kullanici_id INT AUTO_INCREMENT PRIMARY KEY,
    kullanici_adi VARCHAR(50) NOT NULL,
    sifre VARCHAR(50) NOT NULL
);


CREATE TABLE sahalar (
    saha_id INT AUTO_INCREMENT PRIMARY KEY,
    saha_adi VARCHAR(30) NOT NULL,
    sabit_ucret DECIMAL(10,2) NOT NULL DEFAULT 3500.00
);


CREATE TABLE rezervasyonlar (
    rezervasyon_id INT AUTO_INCREMENT PRIMARY KEY,
    musteri_ad_soyad VARCHAR(100) NOT NULL,
    telefon VARCHAR(15),
    saha_adi VARCHAR(30) NOT NULL,
    tarih VARCHAR(20) NOT NULL,
    saat VARCHAR(20) NOT NULL,
    ucret DECIMAL(10,2) NOT NULL DEFAULT 3500.00,
    odendi_mi VARCHAR(10) NOT NULL DEFAULT 'Ödenmedi'
);
